\begin{frame}{¿Qué es Google Analytics?}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Definición de trabajo}
    \small Servicio gratuito de analítica web y de aplicaciones: \textbf{registra interacciones} con un sitio o app y las resume en informes. Existe una versión de pago, Analytics 360.
  \end{block}
  \begin{exampleblock}{En términos de la Unidad I}
    \small Es una fuente de \textbf{datos de primera parte} y de \textbf{señal conductual}: lo que la gente \emph{hace} en \emph{tu} sitio.
  \end{exampleblock}
  \column{0.44\textwidth}
  \begin{alertblock}{Lo que NO es}
    \small No dice \emph{por qué} actúan así (señal declarada), ni qué hacen fuera de tu sitio. Complementa, no sustituye, encuestas, escucha social y CRM.
  \end{alertblock}
\end{columns}
\end{frame}

\section[Cómo funciona]{Cómo funciona GA4}

\begin{frame}{Todo es un evento}
\begin{columns}[T]
  \column{0.5\textwidth}
  \begin{block}{Tres piezas}
    \small
    \begin{itemize}
      \item \textbf{Evento:} interacción con nombre (\texttt{page\_view}, \texttt{purchase}).
      \item \textbf{Parámetros:} describen el evento (página, valor, moneda).
      \item \textbf{Propiedades de usuario:} atributos de la persona (p.\,ej., un segmento).
    \end{itemize}
  \end{block}
  \column{0.46\textwidth}
  \begin{exampleblock}{Ejemplo}
    \scriptsize
    \texttt{purchase}\\[2pt]
    \hspace{1em}\texttt{transaction\_id = T-0001}\\
    \hspace{1em}\texttt{value = 1250.00}\\
    \hspace{1em}\texttt{currency = MXN}\\
    \hspace{1em}ítem: \texttt{p100001}, licuadora\\[3pt]
    usuario: \texttt{segmento\_rfm = Leal}
  \end{exampleblock}
\end{columns}
\vspace{4pt}
\begin{center}\small La pregunta de la Sesión 2 reaparece: \textbf{¿qué comportamiento decides medir?}\end{center}
\end{frame}

\begin{frame}{Cuatro categorías de eventos}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright}p{2.7cm}>{\raggedright}p{4.6cm}>{\raggedright\arraybackslash}p{5.0cm}@{}}
\toprule
\textbf{Categoría} & \textbf{Qué es} & \textbf{Ejemplos} \\
\midrule
Automáticos & Se registran al instalar la etiqueta o el SDK & \texttt{first\_visit}, \texttt{session\_start}, \texttt{user\_engagement} \\
Medición mejorada & Activos por defecto en sitios web & \texttt{page\_view}, \texttt{scroll}, \texttt{click}, \texttt{file\_download}, \texttt{form\_submit} \\
Recomendados & Nombres sugeridos por Google; hay que implementarlos & \texttt{purchase}, \texttt{add\_to\_cart} \\
Personalizados & Los define la organización & \texttt{compra\_en\_tienda} \\
\bottomrule
\end{tabular}
\end{center}

\end{frame}

\begin{frame}{¿Cómo reconoce GA4 a una persona?}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Identificador de cliente (\texttt{client\_id})}
    \small Se guarda en la cookie \texttt{\_ga}. Caducidad por defecto: \textbf{2 años} (configurable; los navegadores pueden acortarla). Reconoce \textbf{navegadores}, no personas.
  \end{block}
  \begin{exampleblock}{User-ID}
    \small Identificador propio de la organización para usuarios con \textbf{inicio de sesión}. Permite unir dispositivos.
  \end{exampleblock}
  \column{0.44\textwidth}
  \begin{alertblock}{Lo que implica}
    \small Una persona con tres dispositivos y sin sesión iniciada aparece como \textbf{tres usuarios}: es el problema de la \textbf{identidad única} de la omnicanalidad (Sesión 3).
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Dimensiones y métricas}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Dimensión = atributo}
    \small Sirve para \textbf{agrupar}: país, canal de adquisición, página, dispositivo.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Métrica = cantidad}
    \small Lo que se \textbf{mide}: usuarios activos, sesiones, número de eventos, eventos clave, ingresos.
  \end{exampleblock}
\end{columns}
\vspace{6pt}
\begin{itemize}
  \small
  \item Un informe es una \textbf{tabla}: una columna por dimensión, una por métrica.
  \item Las \textbf{dimensiones personalizadas} tienen ámbito de \textbf{evento}, \textbf{usuario} o \textbf{artículo} y deben crearse antes de usarse.
  \item Límites de recolección (propiedad estándar): \textbf{25 parámetros por evento}; nombres de evento y parámetro de \textbf{40 caracteres}; valores de parámetro de 100.
\end{itemize}
\end{frame}

\begin{frame}{Informes estándar: colecciones}

  \begin{block}{Colección \textit{Life cycle}}
    \small Sigue el recorrido del cliente:
    \begin{itemize}
      \scriptsize
      \item \textit{Acquisition}: de dónde llegan.
      \item \textit{Engagement}: qué hacen.

\begin{frame}{Práctica 1: ¿de dónde llega el tráfico?}
\begin{block}{En \textit{Reports $\rightarrow$ Acquisition $\rightarrow$ Traffic acquisition}}
  \small
  \begin{enumerate}
    \item Fija el rango a los \textbf{últimos 28 días}.
    \item Cambia la dimensión principal a \textit{Session source / medium}.
    \item Agrega una \textbf{comparación} (p.\,ej., tipo de dispositivo).
    \item ¿Qué canal tiene más sesiones? ¿Cuál tiene mejor tasa de interacción? ¿Coinciden?
  \end{enumerate}
\end{block}
\begin{alertblock}{Lectura crítica}
  \small Compara con el \textbf{mismo periodo del año anterior}: 28 días no bastan para distinguir tendencia de estacionalidad (Capítulo 4).
\end{alertblock}
\end{frame}

\begin{frame}{Exploraciones: siete técnicas}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright}p{3.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Técnica} & \textbf{Qué responde} \\
\midrule
\textit{Free form} & Tablas cruzadas con barras, líneas, dispersión o mapas \\
\textit{Funnel} & Cuántos avanzan en cada paso y dónde se pierden \\
\textit{Path} & Qué rutas siguen por el sitio o la app \\
\textit{Segment overlap} & Cómo se traslapan segmentos de usuarios \\
\textit{User explorer} & Qué hizo cada usuario de un segmento \\
\textit{Cohort} & Grupos con un atributo común (p.\,ej., fecha de llegada) \\
\textit{User lifetime} & Comportamiento y valor a lo largo de su vida como clientes \\
\bottomrule
\end{tabular}
\end{center}
\fuente{Google, Analytics Help, \enquote{Explorations}.}
\end{frame}

\begin{frame}{Práctica 2: un embudo de compra}
\begin{block}{En \textit{Explore $\rightarrow$ Funnel exploration}}
  \small
  \begin{enumerate}
    \item Define los pasos: \texttt{view\_item} $\rightarrow$ \texttt{add\_to\_cart} $\rightarrow$ \texttt{begin\_checkout} $\rightarrow$ \texttt{purchase}.
    \item Anota el porcentaje que avanza de un paso al siguiente.
    \item Señala el paso con \textbf{mayor abandono}: es la \textbf{fricción} del Capítulo 1.
    \item Compara dos segmentos (móvil frente a escritorio): ¿abandonan en el mismo paso?
  \end{enumerate}
\end{block}
\end{frame}

\begin{frame}{¿Cuánto tiempo se conservan los datos?}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Retención en propiedades estándar}
    \small Datos de usuario y de evento: \textbf{2 meses o 14 meses}.
  \end{block}
  \begin{alertblock}{Qué afecta}
    \small Solo a las \textbf{Exploraciones y los embudos}; los \textbf{informes estándar agregados no} se ven afectados.
  \end{alertblock}
  \column{0.44\textwidth}
  \begin{exampleblock}{Consecuencia}
    \small Una exploración de más de 14 meses \enquote{se corta} aunque el informe estándar muestre todo el periodo. Los datos de edad, género e intereses se conservan solo 2 meses.
  \end{exampleblock}
\end{columns}
\fuente{Google, Analytics Help, \enquote{Data retention}.}
\end{frame}

\section[Sacar datos]{Cómo sacar datos de GA4}
%------------------------------------------------

\begin{frame}{Cuatro vías para llevarse los datos}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright}p{2.5cm}>{\raggedright}p{5.6cm}>{\raggedright\arraybackslash}p{3.7cm}@{}}
\toprule
\textbf{Vía} & \textbf{Qué ofrece} & \textbf{Cuándo usarla} \\
\midrule
Exportar desde una exploración & Sheets, CSV, TSV o PDF con \textit{Export data} & Una tabla puntual para Excel o para la app del Cap.~4 \\
Looker Studio & Tableros conectados a la propiedad & Reportes periódicos para terceros \\
Exportación a BigQuery (nube) & Eventos \textbf{crudos}, uno por fila, en un almacén de Google Cloud & Solo lo mencionamos: lo configura el área de TI y puede costar \\
Conexión automática (API) & Conexión automática con otros programas & Solo lo mencionamos: requiere programar; no la usaremos \\
\bottomrule
\end{tabular}
\end{center}
\fuente{Google, Analytics Help y Google for Developers (API de datos de Google Analytics).}
\end{frame}

\begin{frame}{Aviso: la cuenta de demostración no deja exportar}
\begin{alertblock}{Qué pasa en la demo}
  \small La cuenta de demostración de Google es de \textbf{solo consulta}: ahí \textbf{no se pueden descargar ni exportar datos} (la opción \textit{Export data} no está disponible). Es una restricción de esa cuenta pública, no un error tuyo.
\end{alertblock}
\begin{exampleblock}{En una cuenta propia sí}
  \small En una propiedad propia, con un rol de \textbf{Analista o superior}, sí se podrá exportar a CSV, Sheets o PDF. Por eso las prácticas de esta sesión \textbf{no te piden descargar nada}: se resuelven leyendo y capturando la pantalla.
\end{exampleblock}
\scriptsize Verifica en la ayuda vigente de Google qué rol exige cada exportación.
\end{frame}

\begin{frame}{Práctica 3: leer una serie en pantalla}
\begin{block}{Tendencia y estacionalidad, sin descargar}
  \footnotesize
  \begin{enumerate}
    \item En \textit{Explore}, \textit{Free form}, elige visualización de \textbf{líneas}: dimensión \textbf{semana} (o fecha), métrica \textbf{sesiones}; rango lo más largo posible.
    \item \textbf{Captura la pantalla} de la gráfica.
    \item Con lápiz sobre la captura: marca la \textbf{tendencia} (¿sube, baja, se mantiene?) y los \textbf{picos que se repiten}.
    \item Responde en una frase: ¿es un patrón de \textbf{temporada} o un salto de \textbf{campaña}? ¿Hay \textbf{dos ciclos} para confirmarlo?
  \end{enumerate}
\end{block}
\begin{exampleblock}{Con cuenta propia (opcional)}
  \scriptsize Exporta la serie a CSV, déjala en dos columnas en Excel (fecha y valor) y súbela a \texttt{cap04\_series\_datos\_propios.html} con 52 periodos por ciclo.
\end{exampleblock}
\end{frame}

\begin{frame}{Precauciones con lo que sale de GA4}
\begin{alertblock}{Los datos exportados no son \enquote{la verdad}}
  \small Vienen \textbf{agregados} (no hay filas por persona), pueden tener umbrales de privacidad o muestreo en exploraciones grandes, y dos fuentes (informe, otra exportación) pueden \textbf{no coincidir exactamente}.
\end{alertblock}
\begin{block}{Buena práctica}
  \small Registra siempre \textbf{de qué fuente, con qué filtros y con qué fecha de consulta} salió cada cifra: el mismo criterio de verificación del libro.
\end{block}
\end{frame}

\begin{frame}{Analítica = tratamiento de datos personales}
\begin{alertblock}{La base legal no es opcional}
  Una cookie que identifica un navegador, un User-ID o un CRM unido a la analítica son datos que la \textbf{Sesión 3} sitúa bajo la \textbf{LFPDPPP}. Se necesita \textbf{base legal} y \textbf{propósito declarado}: verifica el texto vigente.
\end{alertblock}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Buenas prácticas}
    \scriptsize
    \begin{itemize}
      \item \textbf{Informar} el uso de cookies y analítica.
      \item Usar \textbf{Consent Mode} según el consentimiento.
      \item \textbf{Minimizar}: recolectar solo lo necesario.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Reglas de Google}
    \scriptsize
    \begin{itemize}
      \item \textbf{Prohibido} subir información personal identificable.
      \item Retención: \textbf{2 o 14 meses} según la necesidad real.
    \end{itemize}
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Síntesis}
\begin{itemize}
  \small
  \item GA4 mide \textbf{eventos} con parámetros; reconoce \textbf{navegadores}, no personas, salvo que haya User-ID.
  \item \textbf{Visualizar:} informes estándar para lo habitual; Exploraciones para preguntas a la medida (embudos, rutas, cohortes).
  \item \textbf{Sacar:} CSV y Looker Studio en cuenta propia (la demo no exporta; BigQuery y API quedan para el área técnica); lo exportado está \textbf{agregado} y hay que documentar su fuente.
  \item La interfaz cambia: se aprende el \textbf{modelo} y se \textbf{verifica la vigencia}.
\end{itemize}
\end{frame}

    \item[b)] \alt<2->{\textcolor{IPNguinda}{\textbf{el evento, con sus parámetros}}}{el evento, con sus parámetros};

  \begin{itemize}

  \end{itemize}
  \item Los datos que se exportan de un informe de GA4 vienen:

\begin{frame}{Glosario de la sesión}
\scriptsize
\begin{description}
  \item[Evento / parámetro] Interacción con nombre / dato que la describe (valor, moneda, página).
  \item[Dimensión / métrica] Atributo para agrupar / cantidad que se mide.
  \item[Flujo de datos] Fuente de la propiedad: sitio web o app.
  \item[Evento clave] Evento marcado como importante para el negocio (antes, \enquote{conversión} en Analytics).
  \item[Exploración] Análisis a la medida: tabla libre, embudo, rutas, cohortes\ldots
  \item[Client ID / User-ID] Identificador del navegador (cookie \texttt{\_ga}) / identificador propio para usuarios con sesión.
  \item[BigQuery] Almacén de datos en la nube al que GA4 exporta eventos crudos.
\end{description}
