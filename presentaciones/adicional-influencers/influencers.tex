\begin{frame}{¿Qué hace que alguien sea \enquote{influencer}?}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Definición de trabajo}
    \small Persona que, por su \textbf{credibilidad en un nicho} y su \textbf{comunidad}, puede modificar las decisiones de otros. \emph{No es lo mismo ser famoso que influir.}
  \end{block}
  \begin{exampleblock}{Las tres palancas}
    \small
    \begin{enumerate}
      \item \textbf{Alcance:} cuántos te ven.
      \item \textbf{Resonancia:} cuánto interactúan.
      \item \textbf{Confianza:} cuánto te creen.
    \end{enumerate}
  \end{exampleblock}
  \column{0.44\textwidth}
  \begin{alertblock}{La clave psicológica}
    \small La audiencia siente una relación \textbf{parasocial}: cercanía de un solo sentido, como con un amigo que no sabe que existes. Por eso una recomendación se siente \enquote{de confianza} y por eso debe ser \textbf{transparente}.
  \end{alertblock}
  \small Conecta con la Unidad I: es una \textbf{señal social} que alimenta el perfil del consumidor.
\end{columns}
\end{frame}

\begin{frame}{El tamaño no lo dice todo: seis perfiles}
\centering\scriptsize Un mismo creador puede ser \textbf{micro por tamaño, experto por rol y afiliado por modelo de pago}.\par\vspace{3pt}
\raggedright
\begin{columns}[T]
  \column{0.32\textwidth}
  \begin{block}{Creador de contenido}\scriptsize Vive de producir formatos propios (video, podcast, serie). La marca \textbf{paga por su audiencia y su estilo}.\end{block}
  \begin{block}{Experto / líder de opinión}\scriptsize Médicos, contadores, chefs, ingenieros. Menos seguidores, \textbf{mucha credibilidad}.\end{block}
  \column{0.32\textwidth}
  \begin{block}{Celebridad}\scriptsize Famosa por otro medio (música, deporte, TV). Alcance enorme, \textbf{nicho difuso}.\end{block}
  \begin{exampleblock}{Creador UGC\footnotemark}\scriptsize Produce videos \enquote{estilo usuario} \textbf{para la cuenta de la marca}; no necesita audiencia propia.\end{exampleblock}
  \column{0.32\textwidth}
  \begin{block}{Embajador / afiliado}\scriptsize Relación larga con una marca; cobra comisión por venta con código o enlace.\end{block}
  \begin{alertblock}{Virtual y B2B}\scriptsize Personajes digitales creados por una marca; y el \enquote{LinkedInfluencer} que posiciona a profesionales y empresas.\end{alertblock}
\end{columns}
\footnotetext{\tiny\textbf{UGC} = \emph{User-Generated Content} (contenido generado por usuarios): reseñas, unboxings o videos hechos por personas comunes, con look espontáneo. El \emph{creador UGC} los produce por encargo y la marca los publica en sus propios canales o anuncios; se le paga por el contenido, no por su audiencia.}
\vspace{2pt}
\end{frame}

\begin{frame}{¿Cómo se les paga?}
\begin{columns}[T]
  \column{0.5\textwidth}
  \begin{block}{Cinco modelos comunes}
    \small
    \begin{enumerate}
      \item \textbf{Canje:} producto o experiencia a cambio de contenido.
      \item \textbf{Tarifa fija} por publicación o paquete.
      \item \textbf{Comisión / afiliado:} porcentaje por venta con código o enlace.
      \item \textbf{Embajador:} contrato largo, exclusividad.
      \item \textbf{Ingresos de plataforma:} suscripciones, regalos, monetización.
    \end{enumerate}
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Tarifas por publicación en México (estimación de blog, 2026)}
    \small
    \begin{tabular}{@{}ll@{}}
      Nano  & \$300 – \$2\,000 MXN\\
      Micro & \$2\,000 – \$15\,000\\
      Macro & \$15\,000 – \$120\,000\\
    \end{tabular}\\[3pt]
    \scriptsize Sube entre 20 y 35\% según la misma fuente. Son \textbf{referencias}, no tarifario oficial.
  \end{exampleblock}
\end{columns}
\fuente{Armando Planes (2026), blog; no es una cifra oficial.}
\end{frame}

\begin{frame}{El mercado mexicano: lo que sí y lo que no sabemos}
\begin{columns}[T]
  \column{0.5\textwidth}
  \begin{block}{Señales de crecimiento}
    \small
    \begin{itemize}
      \item IAB México: \textbf{3.3 millones} de creadores de contenido, con 87\% de crecimiento en 2023.
      \item Encuesta global SAMY (vía IAB): \textbf{48\%} de marketers aumentará su inversión; \textbf{45\%} planea más colaboraciones con micro.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{alertblock}{Lo que NO existe todavía}
    \small No hay una \textbf{cifra oficial} de inversión en influencers en México. El estudio de Valor Total Media agrupa creadores con clasificados y otros formatos; AVE, CiM e IAB prevén separar el dato durante 2027.
  \end{alertblock}
\end{columns}
\vspace{4pt}
\begin{center}\small \textbf{Ejercicio de criterio:} ¿por qué un número sin metodología ni fecha es riesgoso para una decisión de inversión?\end{center}

\begin{frame}{Cuatro riesgos que debes saber ver}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{alertblock}{1. Audiencia inflada}
    \small Seguidores comprados, bots y \emph{pods} de comentarios. Señal: engagement \textbf{muy superior a 10\%} en Instagram o comentarios genéricos repetidos.
  \end{alertblock}
  \begin{block}{2. Audiencia equivocada}
    \small Muchos seguidores, pero de otro país o edad. Pide \textbf{demografía real} de la cuenta.
  \end{block}
  \column{0.48\textwidth}
  \begin{block}{3. Riesgo reputacional}
    \small Una polémica del creador se contagia a la marca (\emph{brand safety}).
  \end{block}
  \begin{alertblock}{4. Publicidad no declarada}
    \small Puede ser una infracción y, además, destruye la confianza que se pagó por tener.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Transparencia en México: ¿qué dice la autoridad?}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Guía de Profeco (DOF, agosto 2023)}
    \small
    \begin{itemize}
      \item La publicidad debe ser \textbf{clara, veraz, comprobable} y no inducir a error (art.\,32, Ley Federal de Protección al Consumidor).
      \item El contenido pagado debe \textbf{identificarse} como publicidad.
      \item El creador puede responder junto con la marca si engaña.
    \end{itemize}
  \end{block}
  \column{0.41\textwidth}
  \begin{alertblock}{Matices}
    \small Es una \textbf{guía}: su fuerza vinculante y quién cuenta como influencer siguen en discusión. Hubo denuncias de asociaciones civiles y casos virales de supuestas multas. Verifica siempre la norma vigente.
  \end{alertblock}
\end{columns}
\vspace{4pt}
\begin{center}\small \textbf{Buena práctica:} etiqueta visible (\enquote{publicidad}), al inicio, en el idioma de la audiencia.\end{center}
\fuente{Profeco / DOF (2023); Expansión; Xataka México. La regulación puede haber cambiado.}
\end{frame}

\begin{frame}{Tres métricas para comparar creadores}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Fórmulas}
    \small
    \[ \scriptsize ER = \frac{\text{likes}+\text{comentarios}+\text{guardados}+\text{compartidos}}{\text{seguidores}}\times 100 \]
    \[ \scriptsize CPE = \frac{\text{costo de la publicación}}{\text{interacciones totales}} \]
    \[ \scriptsize CPM_{seg} = \frac{\text{costo}}{\text{seguidores}}\times 1000 \]
  \end{block}
  \column{0.41\textwidth}
  \begin{exampleblock}{Interpretación}
    \small
    \begin{itemize}
      \item \textbf{ER:} qué tan activa es la audiencia.
      \item \textbf{CPE:} cuánto pagas por una interacción.
      \item \textbf{CPM:} cuánto pagas por cada mil personas de alcance potencial.
    \end{itemize}
  \end{exampleblock}
\end{columns}
\vspace{2pt}
\centering\scriptsize Usa el promedio de las \textbf{últimas 10–12 publicaciones}, no la mejor.
\end{frame}

\begin{frame}{Ejemplo numérico: ¿cuál conviene?}
\centering
\small
\begin{tabular}{@{}lrr@{}}

\midrule

Engagement rate & 3\% & 1\%\\
Interacciones por publicación & 900 & 4\,000\\
Tarifa & \$5\,000 & \$60\,000\\

\begin{alertblock}{La respuesta depende del objetivo}
  \small A sale más barato por \emph{interacción}; B por \emph{alcance potencial}. ¿Notoriedad o conversión? Primero el objetivo, luego el creador.
\end{alertblock}

\end{frame}

  \column{0.55\textwidth}
  \begin{block}{Auditoría express de un creador}
    \small
    \begin{enumerate}
      \item Elige a un creador real de tu interés (no una celebridad).
      \item \textbf{Clasifícalo:} nivel, rol, plataforma.
      \item Toma sus \textbf{últimas 10 publicaciones} y calcula su ER promedio.
      \item Busca \textbf{3 señales de alerta} o de calidad en los comentarios.
      \item Propón para qué tipo de marca y objetivo sería ideal.
    \end{enumerate}
  \end{block}
  \column{0.41\textwidth}
  \begin{exampleblock}{Entrega}
    \small Una lámina o un PDF de una cuartilla, en tu carpeta del \textbf{portafolio de evidencias} en Drive, dentro del tiempo límite indicado en clase.
  \end{exampleblock}
  \begin{alertblock}{Ética}
    \small Usa solo \textbf{datos públicos} y no hagas juicios personales sobre el creador.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Cinco ideas para llevarte}
\begin{enumerate}
  \item \textbf{Influir no es ser famoso:} es credibilidad en un nicho.
  \vspace{3pt}
  \item El \textbf{tamaño} (nano a mega) cambia alcance, cercanía y costo; el \textbf{rol} y la \textbf{plataforma} importan igual.
  \vspace{3pt}
  \item Detrás de una publicación hay \textbf{trabajo, contrato y negocio}.
  \vspace{3pt}
  \item Elige con datos: \textbf{ER, CPE y CPM}, según tu objetivo.
  \vspace{3pt}
  \item Sin \textbf{transparencia} no hay confianza, y sin confianza no hay influencia.
\end{enumerate}
\end{frame}
