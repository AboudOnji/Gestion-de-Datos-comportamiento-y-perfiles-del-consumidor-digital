\begin{frame}{Qué es un modelo y por qué hay nueve}
\begin{block}{Modelo}
  \small Representación simplificada que destaca \textbf{algunas} causas de un fenómeno y deja las demás fuera. Ninguno es \enquote{el correcto}: cada uno ilumina una parte de la decisión y oscurece otra.
\end{block}
\begin{alertblock}{Tres cosas que hay que saber desde el inicio}
  \small
  \begin{enumerate}
    \item \textbf{La mayoría no nació como teoría del consumidor}: Marshall escribió economía; Pavlov, fisiología; Veblen, sociología; Freud, psicología clínica. La lectura \enquote{de mercadotecnia} es una \textbf{reconstrucción posterior de los manuales}.
    \item \textbf{No son del mismo tipo}: unos explican \textit{por qué}, otros \textit{cómo se aprende}, otros \textit{predicen}, otro integra variables y uno es un \textbf{método}.
    \item \textbf{Se complementan}: cada uno responde bien una pregunta y mal otras.
  \end{enumerate}
\end{alertblock}
\end{frame}

\section[Clásicos]{3.2.1--3.2.4 Modelos clásicos}

\begin{frame}{3.2.1 Marshall (1890): el modelo económico}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{Idea}
    \small La persona busca la mayor \textbf{satisfacción} (utilidad) con un presupuesto \textbf{limitado}. Cada unidad adicional da menos que la anterior: \textbf{utilidad marginal decreciente}. Reparte su gasto hasta que el último peso rinde lo mismo en todos los usos.
  \end{block}
  \begin{exampleblock}{Excedente del consumidor}
    \small Diferencia entre lo que estaría dispuesta a pagar y lo que paga.
  \end{exampleblock}
  \column{0.46\textwidth}
  \begin{alertblock}{Lente sobre Camila}
    \small Compara tres pares por \textbf{precio y batería} y elige la mejor relación calidad-precio.\\
    \textit{Dato:} uso de comparadores y filtros de precio.
  \end{alertblock}
  \begin{block}{Dónde falla}
    \footnotesize Racionalidad limitada (Simon, 1955); heurísticas y sesgos (Tversky y Kahneman, 1974); no incluye variables sociales ni emocionales. Su lectura digital es \textbf{extrapolación de manuales}.
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{Marshall: dos ejemplos con números}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{exampleblock}{La tercera taza}
    \footnotesize Satisfacción por taza: 10, 6 y 2. Cada taza cuesta 5 (misma escala).\\[2pt]
    Compra la 1.ª ($10>5$) y la 2.ª ($6>5$), \textbf{no la 3.ª} ($2<5$).\\[2pt]
    \textbf{Excedente} $=(10-5)+(6-5)=6$.\\[2pt]
    Un cupón \enquote{2 por 1} funciona porque la 2.ª unidad aún tiene utilidad por encima del precio.
  \end{exampleblock}
  \column{0.48\textwidth}
  \begin{block}{Elasticidad de una suscripción (ficticia)}
    \footnotesize Precio: 149 $\to$ 179 pesos ($+20.1$\,\%). Suscriptores: 10\,000 $\to$ 9\,000 ($-10$\,\%).
    \[ \varepsilon=\frac{-0.100}{+0.201}\approx-0.50 \]
    $|\varepsilon|<1$: demanda \textbf{inelástica}. El ingreso \textbf{sube}: 1\,490\,000 $\to$ 1\,611\,000 pesos al mes.\\[2pt]
    Con $|\varepsilon|>1$ habría bajado.
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{3.2.2 Pavlov (ed. inglesa 1927): estímulo-respuesta}
\begin{columns}[T]
  \column{0.46\textwidth}
  \begin{block}{Idea}
    \footnotesize \textbf{Aprendizaje por asociación}: si un estímulo neutro (EC) se presenta junto a otro que ya provoca una respuesta (EI), termina provocándola solo (RC). Si deja de acompañarse, la respuesta se apaga: \textbf{extinción}.
  \end{block}
  \begin{alertblock}{Lente sobre Camila}
    \footnotesize Los audífonos de cada video de su creadora favorita (música y ambiente que le gustan) acumulan \textbf{afecto por repetición}, antes de evaluarlos.
  \end{alertblock}
  \column{0.50\textwidth}
  \begin{exampleblock}{Ejemplo: sonido de la app de ofertas}
    \scriptsize Sonido (EC) + oferta real (EI) $\to$ emoción (RI). Con el tiempo, el sonido solo (RC) provoca \textbf{anticipación}. Si manda el sonido \textbf{sin oferta}, la asociación se debilita. Si lo asocia con \textbf{urgencia falsa} (\enquote{¡solo quedan 2!}), usa el mismo mecanismo para empujar compras: la diferencia es \textbf{ética}, no técnica.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Pavlov: dónde falla}
\begin{block}{Límites}
  \small
  \begin{itemize}
    \item Subestima \textbf{cognición, atención y contexto}.
    \item Resultados con consumidores \textbf{mixtos}; el efecto se extingue si la asociación no se refuerza.
    \item Es la base conceptual del \textit{retargeting}, pero su eficacia digital está \textbf{menos probada} que la retórica de la industria.
    \item 1927 es la edición inglesa, \textbf{no} el año del descubrimiento.
  \end{itemize}
\end{block}
\end{frame}

\begin{frame}{3.2.3 Veblen (1899): el modelo sociológico}
\begin{columns}[T]
  \column{0.46\textwidth}
  \begin{block}{Idea}
    \small Consumir \textbf{comunica posición social}. \textbf{Consumo conspicuo}: gasto visible cuyo propósito es mostrar estatus. \textbf{Emulación pecuniaria}: cada grupo imita al de arriba.
  \end{block}
  \begin{alertblock}{Lente sobre Camila}
    \small Los audífonos de la marca que usan las personas que admira son más atractivos \textbf{porque son visibles}.
  \end{alertblock}
  \column{0.50\textwidth}
  \begin{exampleblock}{Ejemplo: la edición limitada}
    \footnotesize Tenis de 4\,800 pesos frente a 1\,900 del modelo básico, con poca diferencia funcional. Quien los compra los publica con la etiqueta de la tienda, un \textit{unboxing} y la fila. La publicación es \textbf{parte del producto}. Su demanda puede \textbf{no bajar} cuando sube el precio: el precio alto es parte del mensaje.\\[2pt]
    \textit{Datos:} compras compartidas, interacciones por \enquote{lo más caro}, búsqueda de réplicas.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Veblen: dónde falla}
\begin{block}{Límites}
  \small
  \begin{itemize}
    \item Es un \textbf{ensayo} sobre la clase alta estadounidense de fines del siglo XIX, no un estudio empírico de compra individual.
    \item \enquote{Veblen predijo Instagram} es una \textbf{lectura retrospectiva}; su texto no lo afirma.
    \item No todo el consumo de estatus es visible: existe el consumo de estatus \textbf{silencioso}.
  \end{itemize}
\end{block}
\end{frame}

\begin{frame}{3.2.4 Freud (vía Dichter): corrección conceptual}
\begin{alertblock}{Freud \textbf{no} escribió sobre comportamiento del consumidor}
  \small El modelo llega a la mercadotecnia por \textbf{Ernest Dichter}, que trasladó ideas psicoanalíticas (motivos inconscientes, simbolismo del producto) a la \textbf{investigación motivacional}, popularizada en los años cincuenta (Instituto fundado en 1946).\\[3pt]
  \textbf{Cadena:} psicoanálisis clínico $\rightarrow$ Dichter $\rightarrow$ investigación motivacional.\\[3pt]
  La historiografía es cautelosa con presentar a Dichter como \textbf{único} origen.
\end{alertblock}
\begin{block}{Idea}
  \small Parte de las compras responde a motivos que la persona \textbf{no reconoce o no puede decir} en una encuesta directa. Los productos funcionan como \textbf{símbolos}.
\end{block}
\end{frame}

\begin{frame}{Freud/Dichter: ejemplo, lente y límite}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{exampleblock}{Ejemplo: la app de pagos}
    \footnotesize Encuesta: \enquote{la uso porque es \textbf{rápida}}. Entrevista en profundidad con técnica proyectiva (\enquote{la persona que gasta sin controlar es\ldots}): aparece el \textbf{miedo a perder el control} del dinero. La app que muestra un resumen semanal del gasto responde a ese motivo, que ningún cuestionario cerrado recogió.
  \end{exampleblock}
  \begin{alertblock}{Lente sobre Camila}
    \small La cancelación de ruido puede significar \enquote{\textbf{mi espacio}}: un refugio, algo que no diría en un formulario.
  \end{alertblock}
  \column{0.46\textwidth}
  \begin{block}{Dónde falla}
    \footnotesize Difícil de replicar y validar estadísticamente; su vínculo con el psicoanálisis se ha criticado como \textbf{oportunismo comercial}.
  \end{block}
  \begin{block}{Legado vigente}
    \footnotesize Técnicas cualitativas: entrevistas en profundidad de UX, \textit{insight} de marca. Es una \textbf{continuidad metodológica}, no una cita de Dichter sobre medios digitales.
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{3.2.5 Fishbein y Ajzen (1975, 1991): acción razonada}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Idea}
    \footnotesize La mejor predicción de una conducta es la \textbf{intención}, que depende de dos factores: la \textbf{actitud} y la \textbf{norma subjetiva} (presión social percibida).
  \end{block}
  \begin{block}{Ampliación (Ajzen, 1991)}
    \footnotesize Agrega el \textbf{control conductual percibido}: la facilidad o dificultad que la persona ve en la conducta. Es la \textbf{teoría de la conducta planeada}; predice mejor cuando la persona no controla del todo la conducta (Madden et al., 1992).
  \end{block}
  \column{0.44\textwidth}
  \begin{alertblock}{Cadena del modelo}
    \footnotesize
    Creencias y evaluaciones $\to$ \textbf{actitud}\\
    Creencias normativas y motivación $\to$ \textbf{norma}\\
    Actitud + norma (+ control) $\to$ \textbf{intención} $\to$ conducta
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Fishbein y Ajzen: las fórmulas}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Actitud multiatributo}
    \small $A=\sum_i b_i\,e_i$\\[2pt]
    $b_i$: \textbf{creencia} (qué tan probable es que la marca tenga el atributo).\\
    $e_i$: \textbf{evaluación} (qué tan bueno es ese atributo para la persona).
  \end{block}
  \begin{block}{Norma subjetiva}
    \small $SN=\sum_k nb_k\,mc_k$\\[2pt]
    $nb_k$: cuánto aprueba el referente $k$ la conducta.\\
    $mc_k$: motivación para complacerlo.
  \end{block}
  \column{0.44\textwidth}
  \begin{exampleblock}{Intención}
    \small $BI=w_A\,A+w_N\,SN$\\[2pt]
    Los pesos $w_A, w_N$ se \textbf{estiman con datos}. Para poder sumar, $A$ y $SN$ se reescalan a $[-1,1]$.
  \end{exampleblock}
  \begin{alertblock}{Nota}
    \footnotesize Es de los pocos modelos con programa de investigación empírico extenso y activo; el modelo de aceptación de tecnología (TAM) se deriva de la acción razonada.
  \end{alertblock}
\end{columns}
\end{frame}

  \begin{alertblock}{Lecciones}
    \small
    \begin{itemize}
      \item Una elección clara por actitud puede \textbf{cambiar por presión social}.
      \item Los márgenes pueden ser de \textbf{milésimas}.
      \item Hay un \textbf{peso de la norma} ($\approx0.34$) en que la elección pasa de B a C: reportarlo es más honesto que un ganador único.
    \end{itemize}
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Fishbein y Ajzen: lente y límite}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{alertblock}{Lente sobre Camila}
    \small Pregunta qué piensan sus amistades (\textbf{norma}), evalúa atributos (\textbf{actitud}) y se sabe capaz de pagar (\textbf{control percibido}).
  \end{alertblock}
  \column{0.48\textwidth}
  \begin{block}{Dónde falla}
    \small Supone \textbf{procesamiento deliberado}: subestima hábitos, emociones y procesos automáticos. La \textbf{brecha intención-conducta}: la intención declarada en una encuesta no siempre llega al carrito, y el carrito no siempre llega al pago.
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{3.2.6 O'Shaughnessy (1987): el modelo funcional}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{block}{Idea}
    \footnotesize Crítica de los modelos \enquote{mecanicistas} y de la elección racional. La razón declarada suele ser una \textbf{justificación posterior} de una decisión tomada por vía emocional; \textbf{deseos} y \textbf{necesidades} son cosas distintas; a veces no sabemos qué deseamos hasta que un estímulo nos lo recuerda.
  \end{block}
  \begin{exampleblock}{Ejemplo: el reloj inteligente}
    \scriptsize Dice que lo compró \enquote{para monitorear mi salud}; en entrevista cuenta que su \textbf{grupo de corredores} lo usa y se sentía fuera. \textit{Datos:} reseñas que hablan de comunidad; devoluciones de quienes lo compraron \enquote{por la función}; uso real contra lo dicho.
  \end{exampleblock}
  \column{0.44\textwidth}
  \begin{center}
  \footnotesize
  \begin{tabular}{@{}lll@{}}
  \toprule
   & \textbf{Necesita} & \textbf{No necesita} \\
  \midrule
  \textbf{Desea} & Laptop & 5.ª bocina \\
  \textbf{No desea} & Seguro médico & 4.º cargador \\
  \bottomrule
  \end{tabular}
  \end{center}
  \vspace{2pt}
  \begin{alertblock}{Lente sobre Camila}
    \footnotesize Dirá que compra los audífonos \enquote{para la biblioteca}. Quizá busca dejar de oír el ruido de casa.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{O'Shaughnessy: dónde falla}
\begin{block}{Límites}
  \small
  \begin{itemize}
    \item Es más \textbf{filosófico e interpretativo} que predictivo; difícil de contrastar empíricamente.
    \item No se localizó literatura que lo aplique de forma nombrada a entornos digitales.
    \item Su relevancia aquí es una \textbf{advertencia conceptual}: un clic o una conversión no revelan la motivación detrás. Es una extrapolación pedagógica, no un hallazgo empírico.
  \end{itemize}
\end{block}
\end{frame}

\begin{frame}{3.2.7 Maslow (1943): corrección conceptual}
\begin{alertblock}{La pirámide \textbf{no está} en el artículo original}
  \small El texto de 1943 no contiene ningún diagrama. La pirámide es una \textbf{simplificación posterior}, atribuida de forma habitual a un gráfico de 1960 hecho por otra persona.\\[3pt]
  Además, Maslow \textbf{no} planteó que cada nivel se satisfaga \textbf{completo} antes de pasar al siguiente: una persona puede tener \textbf{parcialmente} satisfechas necesidades de varios niveles a la vez.
\end{alertblock}
\begin{block}{Idea (artículo de 1943)}
  \small Cinco grupos de necesidades: \textbf{fisiológicas, seguridad, pertenencia y amor, estima y autorrealización}. Las de un nivel inferior tienden a predominar mientras estén poco satisfechas, pero la satisfacción es \textbf{gradual y coexiste} entre niveles.
\end{block}
\end{frame}

Fisiológica & Pedido de comida a domicilio, farmacia en línea & Rapidez, disponibilidad \\
Seguridad & Banca móvil, seguros, copias de seguridad, pago protegido & Confianza, garantías, privacidad \\
Pertenencia & Grupos de mensajería, comunidades, regalos entre amistades & \enquote{Eres parte de\ldots} \\

\bottomrule
\end{tabular}}
\end{center}
\vspace{-2pt}
{\tiny \textbf{Lente sobre Camila:} los audífonos le sirven para varias cosas a la vez: concentrarse (logro), llamar a su grupo (pertenencia), sentir control de su entorno (seguridad). \textbf{Límites:} base empírica débil (observación clínica), secuencia estricta difícil de validar entre culturas; útil como \textbf{heurística} para segmentar mensajes. La correspondencia con compras digitales es una traslación conceptual.}
\end{frame}

\end{center}
\vspace{-4pt}
{\scriptsize Primer intento serio de integrar influencias cognitivas, sociales y de mercadotecnia en un solo esquema. (El programa escribe \enquote{Howarth-Sheth}; la grafía correcta es \textbf{Howard}.)}
\fuente{elaboración propia con base en Howard y Sheth (1969), simplificado.}
\end{frame}

\begin{frame}{Howard y Sheth: ejemplo, lente y límite}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{exampleblock}{La consola de Valeria (Cap. 8), como sistema}
    \footnotesize \textbf{Entradas:} precio y características (producto); videos de la creadora y anuncios (simbólicos); opinión del padre y la tía (sociales). \textbf{Proceso:} presta atención, compara y aprende criterios. \textbf{Salidas:} reconoce la marca, actitud favorable, intención y compra. \textbf{Exógenas:} presupuesto del hogar, fecha de su cumpleaños. Su experiencia \textbf{vuelve} al aprendizaje.
  \end{exampleblock}
  \column{0.44\textwidth}
  \begin{alertblock}{Lente sobre Camila}
    \footnotesize Los \textit{touchpoints} son \textbf{entradas}; la navegación y comparación, \textbf{proceso}; la conversión, \textbf{salida}. Se relaciona con el recorrido (Cap.~7), aunque Howard y Sheth escribieron en 1969, sin comercio electrónico.
  \end{alertblock}
  \begin{block}{Dónde falla}
    \footnotesize Complejo; \textbf{constructos difíciles de medir}; capacidad predictiva cuestionada frente a modelos más simples (acción razonada). Se cita más como marco conceptual que como modelo aplicado.
  \end{block}
\end{columns}
\end{frame}

\section[Netnografía]{3.2.9 Netnografía}
%------------------------------------------------

\begin{frame}{3.2.9 Modelo antropológico-etnográfico: la netnografía}
\begin{alertblock}{Una categoría distinta}
  \footnotesize La netnografía es un \textbf{método} de investigación cualitativa, no un modelo teórico como el de Marshall o el de Maslow. El programa la incluye bajo \enquote{modelos de estudio}; aquí se presenta como lo que es.
\end{alertblock}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{Idea (Kozinets, 2002)}
    \footnotesize Las comunidades y culturas de consumo en línea se estudian con técnicas \textbf{etnográficas adaptadas}: observar foros, redes y comunidades de marca para entender sus \textbf{narrativas, símbolos y significados} compartidos. Es el único de los nueve que \textbf{nació para lo digital}.
  \end{block}
  \column{0.46\textwidth}
  \begin{block}{Dónde falla}
    \footnotesize No generaliza estadísticamente; la interpretación depende del investigador; la comunidad no representa a toda la población; \textbf{dilemas éticos} (consentimiento en datos públicos, anonimato, menores de edad).
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{Netnografía: una comunidad de corredores}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{exampleblock}{Ejemplo (ficticio)}
    \footnotesize Una marca de calzado deportivo quiere entender por qué su producto se discute con pasión en un \textbf{foro público} de corredores. Un investigador \textbf{lee durante semanas} los hilos sin intervenir y registra tres temas que no aparecían en la encuesta: el \textbf{ritual} de estrenar tenis antes de una carrera, la \textbf{desconfianza} hacia reseñas pagadas y un \textbf{lenguaje propio} para las lesiones. Quienes más influyen no son los de más seguidores, sino los que \textbf{responden dudas con experiencia propia} (coincide con el Cap.~8). Se rediseñan el contenido y el servicio posventa.
  \end{exampleblock}
  \column{0.41\textwidth}
  \begin{alertblock}{Lente sobre Camila}
    \small Para entender cómo se \textbf{habla} de audífonos entre personas de su edad, la fuente son los \textbf{comentarios y las comunidades}, más que el formulario.
  \end{alertblock}
  \begin{block}{Precaución ética}
    \footnotesize Que una publicación sea pública \textbf{no significa} que la persona haya consentido ser analizada.
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{Tres reglas para elegir}
\begin{enumerate}
  \item \textbf{Empiecen por la pregunta, no por el modelo.}
  \vspace{4pt}
  \item Un modelo que \textbf{predice} (Fishbein y Ajzen) sirve para \textbf{estimar}; uno que \textbf{interpreta} (Dichter, O'Shaughnessy, netnografía) sirve para \textbf{entender}. No pidan predicciones a una interpretación ni significados a una regresión.
  \vspace{4pt}
  \item \textbf{Combinen al menos dos} y dejen por escrito \textbf{qué no puede explicar cada uno}.
\end{enumerate}
\vspace{6pt}
\begin{exampleblock}{Conexión con la Sesión 9A}
  \small El \textbf{tipo de compra} orienta qué modelo conviene: la compra \textbf{compleja} se presta a Fishbein-Ajzen y Howard-Sheth; la \textbf{habitual}, a Pavlov y al hábito; la \textbf{variada}, a la cultura (netnografía).
\end{exampleblock}
\end{frame}

\begin{frame}{Síntesis: nueve lentes, ninguna completa}
\begin{itemize}
  \footnotesize
  \item \textbf{Nueve modelos de orígenes diversos}: Marshall (utilidad), Pavlov (asociación), Veblen (estatus), Freud vía Dichter (motivos inconscientes), Fishbein y Ajzen (actitud y norma), O'Shaughnessy (razones y deseos), Maslow (necesidades), Howard y Sheth (sistema) y la netnografía (método).
  \item \textbf{Dos exigen corrección} sobre lo que dicen los manuales (Freud y Maslow); \textbf{uno es método} y no teoría; y \textbf{todos son extrapolaciones} cuando se aplican a lo digital.
  \item Los datos de comportamiento \textbf{no distinguen} entre modelos: qué se hizo no dice por qué.
  \item \textbf{La pregunta de mercadotecnia decide el modelo}; combinar al menos dos y declarar sus límites es la práctica honesta.
\end{itemize}
\vspace{2pt}
\begin{alertblock}{La regla de la sesión}
  \small Antes de citar un modelo, pregunten: \textbf{¿qué dijo realmente el autor y qué le atribuyen los manuales?}
\end{alertblock}
\end{frame}

\begin{description}
  \item[Actitud multiatributo] Suma de creencia por evaluación sobre los atributos de una marca.
  \item[Norma subjetiva] Presión social percibida: suma de creencia normativa por motivación para complacer.
  \item[Intención] Disposición a realizar una conducta; en Fishbein y Ajzen, combinación ponderada de actitud y norma.
  \item[Control conductual percibido] Facilidad o dificultad que la persona ve en realizar la conducta.
  \item[Utilidad marginal] Satisfacción que aporta la última unidad consumida.
  \item[Elasticidad precio] Cambio porcentual de la cantidad demandada entre el cambio porcentual del precio.
  \item[Extinción] Pérdida de una respuesta condicionada cuando el estímulo condicionado deja de acompañarse del incondicionado.
  \item[Consumo conspicuo] Gasto visible cuyo propósito es mostrar estatus (Veblen).
  \item[Netnografía] Método cualitativo para estudiar comunidades y culturas de consumo en línea.
\end{description}
\end{frame}
