\section[Dos ejes]{3.1 Los dos ejes de la tipología}
%------------------------------------------------

\begin{frame}{Una tipología con dos preguntas}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{¿Cuánto le importa la compra?}
    \small \textbf{Involucramiento} alto o bajo.
  \end{block}
  \column{0.48\textwidth}
  \begin{block}{¿Cuánto se distinguen las marcas?}
    \small \textbf{Diferencias percibidas} grandes o pequeñas.
  \end{block}
\end{columns}
\vspace{6pt}
\begin{itemize}
  \small
  \item Dos ejes $\times$ dos niveles $=$ \textbf{cuatro tipos}.
  \item La tipología se atribuye a \textbf{Henry Assael} (1987) y los manuales de mercadotecnia la reproducen como matriz $2\times2$.
  \item Describe \textbf{cómo} se compra (mucha o poca deliberación). Para el \textbf{por qué} están los modelos de la Sesión 9B.
\end{itemize}
\end{frame}

\begin{frame}{Involucramiento}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{Definición}
    \small Importancia que una persona le da a una \textbf{compra concreta}, medida por cuánto \textbf{esfuerzo} está dispuesta a invertir en decidir.
  \end{block}
  \column{0.46\textwidth}
  \begin{alertblock}{Sube con\ldots}
    \small
    \begin{itemize}
      \item El \textbf{riesgo percibido}: financiero, social, de desempeño.
      \item La \textbf{baja frecuencia} de la compra.
      \item Su relación con la \textbf{identidad} de la persona.
      \item El \textbf{costo de equivocarse}.
    \end{itemize}
  \end{alertblock}
\end{columns}
\begin{exampleblock}{Pregunta de control}
  \footnotesize ¿Qué dos factores suben en la compra de una laptop? ¿Y en la de un detergente?
\end{exampleblock}
\end{frame}

\begin{frame}{Diferencias percibidas entre marcas}
\begin{block}{Definición}
  \small Cuánto distingue la persona unas opciones de otras \textbf{desde su propia mirada}, no desde la ficha técnica.
\end{block}
\begin{exampleblock}{Ejemplo}
  \small Dos colchones con construcción distinta pueden ser \enquote{lo mismo} para quien no sabe qué comparar. Para esa persona, la diferencia \textbf{no existe}, aunque exista en el laboratorio.
\end{exampleblock}
\begin{alertblock}{Consecuencia para la mercadotecnia}
  \small Una campaña puede \textbf{mover este eje}: enseñar a comparar (subir la diferencia percibida) o simplificar la elección (bajarla). Es parte del trabajo de posicionamiento.
\end{alertblock}
\end{frame}

\section[Cuatro tipos]{3.1.1--3.1.4 Los cuatro tipos}
%------------------------------------------------

\begin{frame}{3.1.1 Complejo: alto involucramiento, diferencias grandes}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Cómo decide}
    \small Siente que hay \textbf{mucho en juego} y que las opciones \textbf{sí son distintas}. Sigue el proceso de cinco etapas del Cap.~6 en su versión completa: forma \textbf{creencias} $\to$ \textbf{actitudes} $\to$ \textbf{elige}.
  \end{block}
  \begin{block}{Otros ejemplos}
    \small Seguro de gastos médicos, posgrado en línea, sistema de punto de venta, un teléfono que cuesta dos quincenas.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{La laptop de Camila}
    \small Doce sesiones en nueve días, tres dispositivos, comparadores, tablas de especificaciones, reseñas en video, \enquote{marca \textit{vs}}, y compra en la \textbf{tercera visita} al mismo producto.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.1 Complejo: qué hacer}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Qué busca la persona}
    \footnotesize \textbf{Información comparable y confiable}: fichas claras, reseñas verificadas, comparar dos modelos lado a lado.
  \end{block}
  \begin{block}{Táctica}
    \footnotesize
    \begin{itemize}
      \item Guías de compra y comparadores honestos.
      \item Chat o atención humana en el momento de duda.
      \item Reseñas visibles \textbf{con sus límites}.
      \item Devolución sencilla que baje el riesgo.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{alertblock}{Error típico de la marca}
    \footnotesize Tratar a esta persona con un anuncio emocional de cinco segundos. Ella está \textbf{buscando razones}, y un anuncio sin información no se las da.
  \end{alertblock}
  \begin{exampleblock}{Firma en los datos}
    \footnotesize Muchas sesiones y dispositivos; páginas de comparación; ciclo largo (días o semanas); búsquedas de marca con \enquote{vs}.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.2 Reductor de disonancia: alto involucramiento, diferencias pequeñas}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Cómo decide}
    \small La compra importa, pero las marcas \textbf{parecen similares}. Decide con rapidez (precio, comodidad) y \textbf{después} puede dudar: la \textbf{disonancia posterior a la compra}. Para reducirla busca información que \textbf{confirme} su decisión.
  \end{block}
  \begin{block}{Otros ejemplos}
    \small Sofá, seguro de auto con precios casi iguales, crédito personal, matrícula entre universidades parecidas.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{El colchón de Camila}
    \small Sesión corta y decidida; compra el miércoles por la tarde; \textbf{esa misma noche} regresa a buscar \enquote{opiniones} de la marca que acaba de comprar y abre el correo de confirmación varias veces.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.2 Reductor de disonancia: qué hacer}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Qué busca la persona}
    \footnotesize \textbf{Confirmación y tranquilidad}: que otras personas hayan elegido lo mismo y estén contentas; garantías; atención si algo sale mal.
  \end{block}
  \begin{block}{Táctica}
    \footnotesize
    \begin{itemize}
      \item Correos \textbf{posteriores} a la compra: qué esperar, cómo cuidar el producto.
      \item Reseñas de personas compradoras; garantía visible.
      \item Contacto humano ante cualquier duda.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Firma en los datos}
    \footnotesize Sesión corta; \textbf{regreso tras la compra}; búsqueda de opiniones \textbf{posterior}; devoluciones dentro del plazo de arrepentimiento (indicador de disonancia no resuelta).
  \end{exampleblock}
  \begin{alertblock}{Reducir disonancia no es manipular}
    \footnotesize Confirmar con \textbf{información veraz} ayuda. Ocultar reseñas negativas o inflar las positivas para que no dude, \textbf{no}.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.3 Habitual: bajo involucramiento, diferencias pequeñas}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Cómo decide}
    \small No busca, no compara y no forma una actitud profunda: compra lo que conoce \textbf{porque está a mano}. Lo que parece \enquote{lealtad} puede ser \textbf{inercia}: nadie la eligió, nadie la defendería si cambiara el estante.
  \end{block}
  \begin{block}{Otros ejemplos}
    \small Recarga de saldo, papel higiénico, leche, la app de transporte que siempre abre primero, la suscripción que se renueva sola.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{El detergente de Camila}
    \small \textbf{Cuarenta segundos} entre abrir la tienda y pagar; una sola página; entrada desde \enquote{volver a pedir}; cero búsquedas de marca; una recompra cada cinco o seis semanas con regularidad casi mecánica.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.3 Habitual: qué hacer}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Qué busca la persona}
    \small Que \textbf{no cambie nada}: el mismo producto, en el mismo lugar, con el menor esfuerzo.
  \end{block}
  \begin{block}{Táctica}
    \small
    \begin{itemize}
      \item Reducir fricción: \textbf{recompra de un toque}, suscripción con aviso.
      \item El \enquote{predeterminado} y la disponibilidad.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{alertblock}{Cuidado con la \enquote{lealtad}}
    \small El \enquote{cliente fiel} de la analítica puede desaparecer si un competidor \textbf{aparece primero} en la lista del asistente de compra.
  \end{alertblock}
  \begin{exampleblock}{Firma en los datos}
    \small Una página; entrada directa; recompra regular; cero búsqueda de marca.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.4 Variado: bajo involucramiento, diferencias grandes}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Cómo decide}
    \small No le importa mucho el producto, pero las marcas \textbf{sí son distintas}. Cambia de marca con frecuencia, \textbf{no por insatisfacción} sino por variedad: probar, curiosidad, aburrimiento. Es el tipo más difícil de \enquote{fidelizar}.
  \end{block}
  \begin{block}{Otros ejemplos}
    \small Refresco o bebida energética, música que se descubre en una plataforma, una serie elegida por la portada, un videojuego móvil gratuito.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Las papas de Camila}
    \small Compras frecuentes de ticket bajo con una marca distinta cada vez; clic en promociones de \textbf{edición limitada}; casi no lee reseñas; \textbf{cambio de marca sin queja previa}.
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{3.1.4 Variado: qué hacer}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Qué busca la persona}
    \small \textbf{Algo nuevo}, sin complicaciones.
  \end{block}
  \begin{block}{Táctica según la posición de la marca}
    \small
    \begin{itemize}
      \item \textbf{Líder de categoría:} ocupar el estante digital con variedad propia (sabores, ediciones limitadas, paquetes mixtos), no con descuentos.
      \item \textbf{Retadora:} muestras, tamaños de prueba y novedad en las primeras posiciones.
    \end{itemize}
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Firma en los datos}
    \small Cambio de marca \textbf{sin queja previa}; clic en novedades; ticket bajo; poca lectura de reseñas.
  \end{exampleblock}
  \begin{alertblock}{Atención}
    \small Un descuento \textbf{no} compra lealtad aquí: la persona se va con la siguiente novedad.
  \end{alertblock}
\end{columns}
\end{frame}

\section[Matices]{Lo que la matriz no dice}
%------------------------------------------------

\begin{frame}{Tres aclaraciones para usar bien la matriz}
\begin{enumerate}
  \small
  \item \textbf{El tipo no es del producto: es del episodio.} El detergente es habitual para Camila y puede ser complejo para quien tiene piel alérgica y revisa ingredientes. La primera laptop fue compleja; la segunda, de la misma marca tres años después, será habitual.
  \vspace{3pt}
  \item \textbf{Es una tipología, no una ley.} Las personas se reparten en un \textbf{continuo}; los umbrales entre \enquote{alto} y \enquote{bajo} son una convención del analista y cambian los porcentajes cuando se mueven.
  \vspace{3pt}
  \item \textbf{Hay otra manera de cortar el mismo fenómeno.} Solomon (2017) trata la complejidad de la decisión como un continuo de habitual a extensa, con la decisión limitada en medio. No es contradicción: es otro corte.
\end{enumerate}
\end{frame}

\begin{frame}{Mismo producto, dos episodios}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{Audífonos, primera compra}
    \small Alto involucramiento, diferencias percibidas grandes: \textbf{complejo}. Reseñas, comparadores, consulta con amistades.
  \end{block}
  \column{0.48\textwidth}
  \begin{exampleblock}{Audífonos, reposición idéntica}
    \small Se perdió uno: la misma marca, mismo modelo, un clic. Bajo involucramiento, diferencias pequeñas: \textbf{habitual}.
  \end{exampleblock}
\end{columns}
\vspace{6pt}
\begin{alertblock}{Para la analítica}
  \small Una campaña pensada para \enquote{el comprador de audífonos} le habla a \textbf{una persona que no existe}. Se segmenta por \textbf{episodio}, no por categoría.
\end{alertblock}
\end{frame}

\section[Digital]{El entorno digital mueve el tipo}
%------------------------------------------------

\begin{frame}{La interfaz puede mover una compra de cuadrante}
\footnotesize
\begin{itemize}
  \item Un \textbf{comparador} que ordena por precio y vuelve comparables las opciones \textbf{baja la diferencia percibida}: empuja compras complejas hacia la disonancia (se elige por precio; la duda llega después).
    \item El \textbf{pago de un toque} y la recompra automática \textbf{bajan el involucramiento}: lo que antes se pensaba pasa a ser habitual.
    \item Un asistente que recomienda \textbf{una sola opción} hace lo mismo y concentra la elección en quien programó la recomendación.
    \item El lanzamiento constante de novedades \textbf{sube la diferencia percibida}: lo habitual se vuelve variado.
\end{itemize}
\begin{alertblock}{Idea central}
  \small El tipo no es solo algo que la persona \enquote{trae}: es también algo que el \textbf{diseño de la plataforma produce}.
\end{alertblock}
\end{frame}

\begin{frame}{Frontera (vigente a noviembre de 2026): cuando compra un agente}
\begin{block}{¿Quién tiene el involucramiento?}
  \small Asistentes basados en modelos de lenguaje buscan, comparan y, en algunos casos, \textbf{ejecutan compras} en nombre de la persona (Cap.~4). Si la persona delega una compra de bajo involucramiento, el comportamiento que se observa en los datos es el del \textbf{agente}, no el de la persona.
\end{block}
\begin{alertblock}{Pregunta metodológica estable}
  \small Antes de clasificar a alguien en un cuadrante: ¿se sabe \textbf{quién (o qué)} hizo la compra?
\end{alertblock}
{\scriptsize Las cifras de adopción de estos asistentes provienen de blogs de la industria y no se reportan como dato verificado.}
\end{frame}

\begin{frame}{Ética: lo habitual es terreno fértil para los patrones oscuros}
\begin{columns}[T]
  \column{0.52\textwidth}
  \begin{alertblock}{Por qué importa}
    \small En la compra habitual \textbf{no hay deliberación}: no hay oportunidad de que la persona se detenga a evaluar. Ahí se concentran las renovaciones automáticas difíciles de cancelar, las casillas premarcadas y los cobros recurrentes que ya nadie recuerda.
  \end{alertblock}
  \column{0.44\textwidth}
  \begin{block}{Cuidados mínimos}
    \small
    \begin{itemize}
      \item \textbf{Consentimiento} claro para cobros recurrentes.
      \item \textbf{Facilidad de salida}: cancelar tan fácil como contratar.
      \item \textbf{Recordatorio} antes de renovar.
      \item Base legal para el tratamiento de datos (LFPDPPP vigente).
    \end{itemize}
  \end{block}
\end{columns}
\vspace{4pt}
{\small\color{IPNgris} Un perfil que \enquote{acierta} sobre una persona también puede exponer algo que ella no eligió mostrar.}
\end{frame}

\begin{frame}{Actividad: clasifica tus compras (20 min)}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Instrucciones}
    \small
    \begin{enumerate}
      \item Retomen las \textbf{cuatro compras} del arranque.
      \item Ubíquenlas en la matriz: ¿involucramiento alto o bajo? ¿diferencias grandes o pequeñas?
      \item Escriban \textbf{un indicio} (tiempo, búsquedas, regreso al sitio) que sostenga cada clasificación.
      \item En parejas: intercambien y \textbf{discutan un desacuerdo}.
    \end{enumerate}
  \end{block}
  \column{0.41\textwidth}
  \begin{exampleblock}{Para discutir}
    \small
    \begin{itemize}
      \item ¿Qué compra \textbf{no cabe} bien en ninguna casilla?
      \item ¿Qué dato \textbf{no tendría} una tienda para clasificarla?
      \item ¿Alguna de sus compras cambió de tipo con el tiempo?
    \end{itemize}
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Caso: Tienda Brisa}
\begin{exampleblock}{Caso ficticio}
  \small Tienda en línea que vende \textbf{audífonos, detergente, botanas y colchones}. A \textbf{todas} las personas clientes les envía el mismo flujo de correos: (1) oferta del 10\,\% por \enquote{terminar tu compra}; (2) cuestionario de satisfacción; (3) \enquote{Productos que te pueden gustar}.
\end{exampleblock}
\begin{alertblock}{Resultados del trimestre}
  \footnotesize La oferta funciona bien con los \textbf{audífonos}, mal con el \textbf{detergente} (la gente no lo había dejado en el carrito: lo compra cada mes). Los \textbf{colchones} tienen la mayor tasa de devolución. Las \textbf{botanas} se venden en olas: lo que gana un mes pierde el siguiente.
\end{alertblock}
\end{frame}

\begin{frame}{Preguntas guía}
\small
\begin{enumerate}
  \item \textbf{Clasifiquen} cada categoría (audífonos, detergente, botanas, colchones) en uno de los cuatro tipos y justifiquen con las \textbf{firmas de datos}.
  \vspace{3pt}
  \item Para cada una, \textbf{rediseñen el flujo de correos}. ¿Qué cambia entre el del colchón y el del detergente?
  \vspace{3pt}
  \item ¿Por qué la oferta de \enquote{terminar tu compra} \textbf{no funciona} con el detergente? Respondan con la tipología.
  \vspace{3pt}
  \item ¿Qué \textbf{riesgo ético} tiene un flujo de recompra automática sin recordatorio para el detergente?
\end{enumerate}
\begin{alertblock}{Nota}
  \small Las preguntas sobre \textbf{modelos de actitud y de precio} del mismo caso se resuelven en la Sesión 9B.
\end{alertblock}
\end{frame}

\begin{frame}{Síntesis: el tipo es del episodio}
\begin{itemize}
  \small
  \item Dos ejes (\textbf{involucramiento} y \textbf{diferencias percibidas}) producen cuatro tipos: \textbf{complejo, reductor de disonancia, habitual y variado}.
  \item El tipo pertenece al \textbf{episodio}, no al producto, y el \textbf{entorno digital} puede moverlo.
  \item Cada tipo deja una \textbf{firma distinta} en los datos y pide una \textbf{táctica distinta}.
  \item Los umbrales entre alto y bajo son \textbf{convención del analista}; las firmas son \textbf{indicios}, no pruebas.
  \item Lo habitual exige especial cuidado ético: \textbf{consentimiento y facilidad de salida}.
\end{itemize}
\vspace{4pt}
\begin{alertblock}{La regla de la sesión}
  \small Antes de elegir una táctica, pregunten: \textbf{¿qué tipo de compra es esta, y qué dato me lo dice?}
\end{alertblock}
\end{frame}

\begin{frame}{Autoevaluación rápida}
\footnotesize
\begin{enumerate}
  \item Compra un sofá caro; todas las marcas le parecen similares y esa noche busca opiniones de la marca elegida. Es comportamiento:
  \begin{itemize}
    \item[a)] complejo; \item[b)] \alt<2->{\textcolor{IPNguinda}{\textbf{reductor de disonancia}}}{reductor de disonancia}; \item[c)] habitual; \item[d)] variado.
  \end{itemize}
  \item Quien cambia de botana en cada compra, sin queja alguna, muestra comportamiento:
  \begin{itemize}
    \item[a)] \alt<2->{\textcolor{IPNguinda}{\textbf{variado}}}{variado}; \item[b)] habitual; \item[c)] complejo; \item[d)] reductor de disonancia.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Autoevaluación rápida (2)}
\footnotesize
\begin{enumerate}
\setcounter{enumi}{2}
  \item El tipo de comportamiento de compra es, sobre todo, propiedad de:
  \begin{itemize}
    \item[a)] el producto; \item[b)] \alt<2->{\textcolor{IPNguinda}{\textbf{el episodio de compra}}}{el episodio de compra}; \item[c)] la marca; \item[d)] el canal.
  \end{itemize}
  \item El pago de un toque tiende a mover una compra hacia:
  \begin{itemize}
    \item[a)] lo complejo; \item[b)] \alt<2->{\textcolor{IPNguinda}{\textbf{lo habitual (baja el involucramiento)}}}{lo habitual (baja el involucramiento)}; \item[c)] lo variado; \item[d)] ninguno.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Glosario de la sesión}
\scriptsize
\begin{description}
  \item[Involucramiento] Importancia que la persona da a una compra, medida por el esfuerzo que invierte en decidir.
  \item[Diferencias percibidas] Cuánto distingue la persona las marcas entre sí, desde su mirada.
  \item[Riesgo percibido] Probabilidad y costo que la persona atribuye a equivocarse: financiero, social o de desempeño.
  \item[Disonancia posterior a la compra] Incomodidad de haber elegido entre opciones poco distintas.
  \item[Inercia] Compra repetida sin elección deliberada; se confunde con lealtad.
  \item[Búsqueda de variedad] Cambio de marca por probar, no por insatisfacción.
  \item[Episodio de compra] Unidad de análisis de la tipología: una compra concreta, de una persona, en un momento.
  \item[Firma en los datos] Patrón observable (tiempo, sesiones, regreso, cambio de marca) que sugiere un tipo.
\end{description}
\end{frame}
